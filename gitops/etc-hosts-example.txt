# /etc/hosts configuration for GitOps Home Infrastructure
#
# Add these entries to access services:
#
# Linux/Mac:
#   sudo vim /etc/hosts
#
# Windows (as Administrator):
#   notepad C:\Windows\System32\drivers\etc\hosts
#
# Replace <SERVER_IP> with your actual server IP

# PRODUCTION CLUSTER - External Access
<SERVER_IP> dashy.local      # Main Dashboard - Gateway to all services
<SERVER_IP> argocd.local     # GitOps Management UI

# DEVELOPMENT CLUSTER - External Access (if running develop cluster)
<SERVER_IP> dashy-dev.local  # Dev Dashboard
<SERVER_IP> argocd-dev.local # Dev GitOps UI
